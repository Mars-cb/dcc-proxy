cmake_minimum_required(VERSION 3.13)
project(dcc-proxy VERSION 0.1.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
include(GNUInstallDirs)
find_package(Qt5 5.11 REQUIRED COMPONENTS Core Widgets Network Qml DBus)
find_package(DtkWidget REQUIRED)
find_path(DCC_INTERFACE_DIR interface/moduleinterface.h
  PATHS /usr/include/dde-control-center
  "${CMAKE_CURRENT_SOURCE_DIR}/third_party/dcc-interface")
if(NOT DCC_INTERFACE_DIR)
  message(FATAL_ERROR "dde-control-center-dev headers are required")
endif()
set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_INSTALL OFF CACHE BOOL "" FORCE)
set(YAML_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/yaml-cpp EXCLUDE_FROM_ALL)
add_library(party-common STATIC src/subscriptionnodes.cpp src/subscriptionnodes.h src/geodata.cpp src/geodata.h src/tun.cpp src/tun.h src/proxybypass.cpp src/config.cpp src/config.h src/api.cpp src/api.h src/nodeflags.cpp src/nodeflags.h)
target_include_directories(party-common PUBLIC src)
target_link_libraries(party-common PUBLIC Qt5::Core Qt5::Network Qt5::DBus yaml-cpp)
target_compile_options(party-common PRIVATE -Wall -Wextra -Werror)
add_library(dcc-proxy MODULE src/plugin.cpp src/plugin.h src/panel.cpp src/panel.h src/nodelist.cpp src/nodelist.h src/resources.qrc)
target_include_directories(dcc-proxy PRIVATE "${DCC_INTERFACE_DIR}")
target_link_libraries(dcc-proxy PRIVATE party-common Qt5::Widgets Dtk::Widget)
target_compile_options(dcc-proxy PRIVATE -Wall -Wextra -Werror)
add_executable(dde-mihomo-daemon src/service.cpp)
target_link_libraries(dde-mihomo-daemon PRIVATE party-common Qt5::Qml)
target_compile_options(dde-mihomo-daemon PRIVATE -Wall -Wextra -Werror)
add_executable(dcc-proxy-preview src/preview.cpp src/panel.cpp src/panel.h src/nodelist.cpp src/nodelist.h src/resources.qrc)
target_link_libraries(dcc-proxy-preview PRIVATE party-common Qt5::Widgets Dtk::Widget)
target_compile_options(dcc-proxy-preview PRIVATE -Wall -Wextra -Werror)
install(TARGETS dcc-proxy LIBRARY DESTINATION lib/dde-control-center/modules)
install(TARGETS dde-mihomo-daemon RUNTIME DESTINATION lib/dcc-proxy)
install(FILES packaging/dde-mihomo-daemon.service DESTINATION lib/systemd/user)
install(FILES README.md DESTINATION share/doc/dcc-proxy)
install(DIRECTORY docs DESTINATION share/doc/dcc-proxy)
install(FILES LICENSE DESTINATION share/doc/dcc-proxy)
install(FILES third_party/yaml-cpp/LICENSE DESTINATION share/doc/dcc-proxy RENAME yaml-cpp.LICENSE)
install(FILES third_party/flag-icons/LICENSE DESTINATION share/doc/dcc-proxy RENAME flag-icons.LICENSE)
set(MIHOMO_BINARY "" CACHE FILEPATH "Optional native Mihomo executable to bundle")
set(MIHOMO_SOURCE_ARCHIVE "" CACHE FILEPATH "Corresponding source archive for bundled Mihomo")
if(MIHOMO_BINARY)
  if(NOT EXISTS "${MIHOMO_SOURCE_ARCHIVE}")
    message(FATAL_ERROR "Bundled core requires MIHOMO_SOURCE_ARCHIVE")
  endif()
  install(PROGRAMS "${MIHOMO_BINARY}" DESTINATION lib/dcc-proxy RENAME mihomo)
  install(FILES "${MIHOMO_SOURCE_ARCHIVE}" DESTINATION share/doc/dcc-proxy RENAME mihomo-source.tar.gz)
  install(FILES packaging/mihomo.LICENSE DESTINATION share/doc/dcc-proxy)
endif()
set(MIHOMO_GEODATA_DIR "" CACHE PATH "Directory containing verified official offline geodata")
if(MIHOMO_GEODATA_DIR)
  foreach(database Country.mmdb GeoIP.dat GeoSite.dat ASN.mmdb)
    if(NOT EXISTS "${MIHOMO_GEODATA_DIR}/${database}")
      message(FATAL_ERROR "Missing bundled ${database}; run tools/prepare-geodata.py")
    endif()
    install(FILES "${MIHOMO_GEODATA_DIR}/${database}" DESTINATION share/dcc-proxy)
  endforeach()
  install(FILES packaging/geodata.json DESTINATION share/dcc-proxy)
  install(FILES packaging/geodata.LICENSE DESTINATION share/doc/dcc-proxy)
  install(FILES packaging/geodata-NOTICES.md DESTINATION share/doc/dcc-proxy)
endif()
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_CONTACT "DCC Clash Party maintainers")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native Clash / Mihomo control center plugin")
set(CPACK_DEBIAN_PACKAGE_SECTION net)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "dde-control-center, libqt5qml5, libqt5svg5")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_CURRENT_SOURCE_DIR}/debian/preinst;${CMAKE_CURRENT_SOURCE_DIR}/debian/postinst")
include(CPack)
if(BUILD_TESTING)
  find_package(Qt5 REQUIRED COMPONENTS Test)
  add_executable(party-tests tests/test_party.cpp src/plugin.cpp src/plugin.h src/panel.cpp src/panel.h src/nodelist.cpp src/nodelist.h src/resources.qrc)
  target_include_directories(party-tests PRIVATE "${DCC_INTERFACE_DIR}")
  target_link_libraries(party-tests PRIVATE party-common Qt5::Test Qt5::Widgets Qt5::Qml Dtk::Widget)
  add_test(NAME party-tests COMMAND party-tests)
  set_tests_properties(party-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
